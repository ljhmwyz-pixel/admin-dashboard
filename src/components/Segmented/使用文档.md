# 分段控制器

## 1. 组件方式

```tsx
/**
 * 状态筛选类型
 */
export type StatusFilter = 'all' | 'normal' | 'deleted';
// 状态筛选：'all' | 'normal' | 'deleted'
const [statusFilter, setStatusFilter] = useState<'all' | 'normal' | 'deleted'>('all');
/**
 * OptionItem接口说明
 * label: React.ReactNode;
 * value: string | number;
 * icon?: React.ReactNode;(未选中的icon)
 * iconSelected?: React.ReactNode(已选中的icon);
 * disabled?: boolean;
 * color?: string; (已选中的颜色)
 */
const statusList: OptionItem[] = [
  { label: 'All', value: 'all', color: '#33C2C8' },
  {
    label: 'Normal',
    value: 'normal',
    color: '#31C47F',
  },
  {
    label: 'Deleted',
    value: 'deleted',
    color: '#F45858',
  },
];
// 状态筛选
const handleStatusChange = useCallback((status: 'all' | 'normal' | 'deleted') => {
  setStatusFilter(status);
}, []);
```
